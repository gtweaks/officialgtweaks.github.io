GTweaks Tweaks Script.

A conservative PowerShell starter script for supported Windows settings.

GTweaks-Optimize.ps1 
# GTweaks Conservative Windows Script

# Review this script before running.
# Run PowerShell as Administrator.

$ErrorActionPreference = "Stop"

Write-Host "GTweaks - Creating restore point..."

try {
    Enable-ComputerRestore -Drive "$env:SystemDrive\"

    Checkpoint-Computer `
        -Description "GTweaks Before Optimization" `
        -RestorePointType "MODIFY_SETTINGS"

} catch {
    Write-Warning "Restore point could not be created automatically."
}

Write-Host "Enabling Windows Game Mode..."

New-Item `
    -Path "HKCU:\Software\Microsoft\GameBar" `
    -Force | Out-Null

Set-ItemProperty `
    -Path "HKCU:\Software\Microsoft\GameBar" `
    -Name "AutoGameModeEnabled" `
    -Type DWord `
    -Value 1

Write-Host "Checking for High Performance power plan..."

$high = (
    powercfg /L |
    Select-String -Pattern "High performance"
)

if ($high) {

    $guid = [regex]::Match(
        $high.ToString(),
        "[0-9a-fA-F-]{36}"
    ).Value

    if ($guid) {
        powercfg /S $guid
    }
}

Write-Host "GTweaks starter optimization complete!" -ForegroundColor Green
Write-Host "Restart Windows if necessary."
